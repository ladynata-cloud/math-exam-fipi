  function countDecodings(encoded, letters, codes, limit=2){
    const memo = new Map();
    function rec(pos){
      if(pos === encoded.length) return [''];
      if(memo.has(pos)) return memo.get(pos);
      let out = [];
      for(const ch of letters){
        const code = codes[ch];
        if(encoded.startsWith(code, pos)){
          for(const tail of rec(pos + code.length)){
            out.push(ch + tail);
            if(out.length >= limit) break;
          }
        }
        if(out.length >= limit) break;
      }
      memo.set(pos, out);
      return out;
    }
    return rec(0);
  }

  function shortestPath(nodes, edges, start, end){
    const adj = Object.fromEntries(nodes.map(n=>[n,[]]));
    edges.forEach(e => { adj[e.u].push([e.v,e.w]); adj[e.v].push([e.u,e.w]); });
    let best = {dist: Infinity, path: []};
    function dfs(u, seen, dist, path){
      if(dist >= best.dist) return;
      if(u === end){ best = {dist, path:path.slice()}; return; }
      for(const [v,w] of adj[u]) if(!seen.has(v)){
        seen.add(v); path.push(v); dfs(v, seen, dist+w, path); path.pop(); seen.delete(v);
      }
    }
    dfs(start, new Set([start]), 0, [start]);
    return Number.isFinite(best.dist) ? best : null;
  }
  function pathWeights(path, edges){
    const out=[];
    for(let i=0;i<path.length-1;i++){
      const a=path[i], b=path[i+1];
      out.push(edges.find(e => (e.u===a&&e.v===b)||(e.u===b&&e.v===a)).w);
    }
    return out;
  }

  function countPathsDAG(nodes, edges, end){
    const adj = Object.fromEntries(nodes.map(n=>[n,[]]));
    edges.forEach(([u,v]) => adj[u].push(v));
    const count = Object.fromEntries(nodes.map(n=>[n,0]));
    count[end] = 1;
    for(let i=nodes.length-2; i>=0; i--){
      const u = nodes[i];
      count[u] = adj[u].reduce((sum,v) => sum + count[v], 0);
    }
    return count;
  }
  function renderDagSvg(nodes, edges, coords){

      const sx=x1+dx/len*pad, sy=y1+dy/len*pad, ex=x2-dx/len*pad, ey=y2-dy/len*pad;
      const curve = Math.abs(y2-y1) > 90 ? `Q ${(sx+ex)/2} ${(sy+ey)/2 + (i%2?18:-18)} ${ex} ${ey}` : `L ${ex} ${ey}`;
      return `<path d="M ${sx} ${sy} ${curve}" stroke="#24324f" stroke-width="2.4" fill="none" marker-end="url(#arrow)"/>`;
    }).join('');
    const nodeSvg = nodes.map(n => {
      const [x,y]=coords[n];
      return `<g><circle cx="${x}" cy="${y}" r="18" fill="#fff" stroke="#3757ff" stroke-width="3"/><text x="${x}" y="${y+6}" text-anchor="middle" font-family="Arial" font-size="17" font-weight="800" fill="#152033">${n}</text></g>`;
    }).join('');
    return `<svg class="graph-svg" viewBox="0 0 630 320" role="img" aria-label="Ориентированный граф">
      <defs><marker id="arrow" markerWidth="10" markerHeight="10" refX="8" refY="3" orient="auto" markerUnits="strokeWidth"><path d="M0,0 L0,6 L9,3 z" fill="#24324f"/></marker></defs>
      ${edgeSvg}${nodeSvg}
    </svg>`;
  }

  function gen10(){
    const bases = [2,8,16];
    const terms = bases.map(base => {
      const val = base === 2 ? rnd(20,120) : base === 8 ? rnd(20,200) : rnd(20,240);
      return {val, base, text: val.toString(base).toUpperCase()};
    });
    const shuffled = shuffle(terms);
    const total = shuffled.reduce((s,t)=>s+t.val,0);
    const expr = shuffled.map(t => `${t.text}<sub>${t.base}</sub>`).join(' + ');
    const conv = shuffled.map(t => `<span class="mono">${t.text}${sub(t.base)}</span> = ${t.val}`).join('; ');
    return {
      num:10,
      title:'Системы счисления',
      answer:String(total),
      displayAnswer:String(total),
      html:`<p>Вычислите значение арифметического выражения:</p>
        <p style="font-size:26px;font-weight:950">${expr}</p>
        <p>В ответе запишите десятичное число. Основание системы счисления указывать не нужно.</p>`,
      hint:'Переведи каждое число в десятичную систему, потом сложи. В шестнадцатеричной системе A=10, B=11, C=12, D=13, E=14, F=15.',
      solution:`Переводим: ${conv}. Сумма: ${shuffled.map(t=>t.val).join(' + ')} = <b>${total}</b>.`
    };
  }

        <div class="mini-head"><span>№${task.num}. ${esc(task.title)}</span><span class="mini-result" id="miniResult${idx}">—</span></div>

        <div class="hint-box" id="miniHint${idx}">${task.hint}</div>
        <div class="solution-box" id="miniSolution${idx}">${task.solution}</div>
      </div>`).join('');

    [...document.querySelectorAll('[data-mini-hint]')].forEach(btn => btn.addEventListener('click', () => $('miniHint'+btn.dataset.miniHint).classList.toggle('show')));

  function checkMini(idx, shouldRecord=false){
    const task = variantTasks[idx];
    const input = $('miniAnswer'+idx).value;
    const ok = checkTask(task, input);
    const badge = $('miniResult'+idx);
    badge.textContent = ok ? 'верно' : `ответ: ${task.displayAnswer ?? task.answer}`;
    badge.className = 'mini-result ' + (ok ? 'ok' : 'no');
    $('miniSolution'+idx).classList.add('show');
    if(shouldRecord) record(task.num, ok);
    return ok;
  }
  function checkVariant(){
    if(!variantTasks.length) return;
    let okCount = 0;
    variantTasks.forEach((_, idx) => { if(checkMini(idx, true)) okCount += 1; });
    $('variantScore').textContent = `${okCount}/10 верно`;
    toast(okCount >= 8 ? `Сильно! ${okCount}/10 — уже пахнет уверенной первой частью.` : `${okCount}/10. Нормальная разминка: посмотри решения и добей слабые номера.`);
  }
  function startVariantTimer(){
    if(variantTimer) clearInterval(variantTimer);
    variantSeconds = 30 * 60;
    updateTimer();
    variantTimer = setInterval(() => {
      variantSeconds -= 1;
      updateTimer();
      if(variantSeconds <= 0){ clearInterval(variantTimer); variantTimer = null; toast('30 минут вышли. Проверь вариант и отметь номера, где зависал дольше всего.'); }
    }, 1000);
  }
  function updateTimer(){
    const s = Math.max(0, variantSeconds);
    const m = Math.floor(s/60), sec = s%60;
    $('variantTimer').textContent = `${String(m).padStart(2,'0')}:${String(sec).padStart(2,'0')}`;
  }

    const t = $('toast'); t.textContent = msg; t.classList.add('show');
    clearTimeout(toast._timer); toast._timer = setTimeout(()=>t.classList.remove('show'), 3600);
  }
